import { afterEach, expect, it, vi } from 'vitest';
import { IDBFactory } from 'fake-indexeddb';
import { WorkStore } from '../src/offline/runtime/store';
import { OfflineEngine } from '../src/offline/runtime/engine';
import { projectCommand } from '../src/offline/runtime/project';
import { offlineSnapshot } from './helpers/offlineSnapshot';
import { loadCurrentCollectionQueue } from '../src/lib/collectionQueue';
const rpc = vi.hoisted(() => vi.fn());
vi.mock('../src/lib/supabase', () => ({supabase: {rpc}}));
afterEach(() => {vi.restoreAllMocks(); vi.useRealTimers();});
it('review: real courier queue shape becomes payment denied despite an open run and collection permission', async () => {
 const s=offlineSnapshot();
 // This shape is verified against all real migrations by the SQL probe.
 rpc.mockResolvedValue({error:null,data:{collection_run_id:'run',status:'open',queue:[]}});
 s.collection=await loadCurrentCollectionQueue(s.serviceDate);
 expect(s.profile.can_collect_shop_payments).toBe(true);
 expect(s.collection.isClosed).toBe(false);
 expect(s.collection.canRecordPayment).toBe(false);
 expect(() => projectCommand(s, {type:'collection_payment',args:{}} as any)).toThrow('บัญชีนี้หรือรอบนี้ยังไม่เปิดรับเงิน');
});
it('review: permission errors remain endlessly retryable and block later commands', async () => {
 vi.useFakeTimers({toFake:['Date']});vi.setSystemTime('2026-10-29T02:00:00Z');
 vi.spyOn(navigator,'onLine','get').mockReturnValue(true);
 const s=offlineSnapshot(),store=new WorkStore(new IDBFactory(),'review');
 const send=vi.fn(async()=>{throw {code:'42501',message:'permission denied for table employee_offline_commands'};});
 const engine=new OfflineEngine(s.ownerId,s.serviceDate,store,{send,resolutions:async()=>[]});
 try {
  await engine.install(s);
  for(let n=0;n<2;n++) await engine.enqueue({id:crypto.randomUUID(),type:'stock_transfer',rpc:'record_employee_stock_transfer',args:{p_round_id:'round',p_items:[{ice_type_id:'ice',quantity:1}]},expected:{},recordedAt:new Date().toISOString(),label:'withdrawal'});
  await engine.sync();vi.setSystemTime(Date.now()+61000);await engine.sync();
  expect(send).toHaveBeenCalledTimes(2);
  expect((await store.commands(s.ownerId)).map(c=>c.status)).toEqual(['retry_wait','pending']);
 }finally{await store.close();}
});
